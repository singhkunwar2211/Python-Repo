from flask import Flask, request, render_template, send_file, redirect, url_for
from scanner import generate_ip_range, scan_ips_parallel
from io import BytesIO, StringIO
import csv
import json

app = Flask(__name__)
scan_cache = []  # To hold last scan results

@app.route("/", methods=["GET", "POST"])
def index():
    global scan_cache
    results = []
    error = None
    if request.method == "POST":
        ip_range = request.form.get("ip_range", "")
        try:
            ip_list = generate_ip_range(ip_range)
            # Reduce max_threads to 10 for better performance
            results = scan_ips_parallel(ip_list, max_threads=10)
            scan_cache = results  # Store in cache
        except Exception as e:
            error = f"Invalid IP range or scanning error: {str(e)}"
    return render_template("index.html", devices=scan_cache, error=error)

@app.route("/download/<filetype>")
def download(filetype):
    if not scan_cache:
        return redirect(url_for('index'))

    if filetype == "csv":
        # CSV must use StringIO (text stream), then convert to BytesIO for send_file
        text_stream = StringIO()
        writer = csv.DictWriter(text_stream, fieldnames=["ip", "mac", "hostname", "vendor", "os"])
        writer.writeheader()
        writer.writerows(scan_cache)
        byte_stream = BytesIO(text_stream.getvalue().encode("utf-8"))
        byte_stream.seek(0)
        return send_file(
            byte_stream,
            mimetype="text/csv",
            as_attachment=True,
            download_name="scan_results.csv"
        )

    elif filetype == "json":
        # JSON can go directly to BytesIO
        json_data = json.dumps(scan_cache, indent=4)
        byte_stream = BytesIO(json_data.encode("utf-8"))
        byte_stream.seek(0)
        return send_file(
            byte_stream,
            mimetype="application/json",
            as_attachment=True,
            download_name="scan_results.json"
        )

    return redirect(url_for('index'))

if __name__ == "__main__":
    app.run(debug=True)

import subprocess
import socket
import platform
import re
import nmap
from concurrent.futures import ThreadPoolExecutor, as_completed

IS_WINDOWS = platform.system().lower() == "windows"

# Function to ping IP
def ping_ip(ip):
    try:
        if IS_WINDOWS:
            subprocess.run(["ping", "-n", "1", "-w", "1000", ip], stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
        else:
            subprocess.run(["ping", "-c", "1", "-W", "1", ip], stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
    except Exception:
        pass

# Function to get MAC address from ARP
def get_mac_from_arp(ip):
    try:
        if IS_WINDOWS:
            arp_output = subprocess.check_output(["arp", "-a"], encoding="utf-8")
        else:
            arp_output = subprocess.check_output(["arp", "-n"], encoding="utf-8")

        mac_regex = re.compile(r"([0-9a-fA-F]{2}[:-]){5}([0-9a-fA-F]{2})")
        for line in arp_output.splitlines():
            if ip in line:
                match = mac_regex.search(line)
                return match.group(0) if match else "Unknown"
    except Exception:
        pass
    return "Unknown"

# Function to resolve hostname from IP
def resolve_hostname(ip):
    try:
        return socket.gethostbyaddr(ip)[0]
    except socket.herror:
        return "Unknown"

# Function to get vendor and OS information using Nmap
def get_vendor_and_os(ip):
    nm = nmap.PortScanner()
    try:
        nm.scan(ip, arguments='-T4 -F -O')  # Adding -O for OS detection
        os = nm[ip].get("osmatch", [{"name": "Unknown"}])[0]['name']
        vendor = nm[ip].get("hostnames", ["Unknown"])[0]
        return vendor, os
    except Exception as e:
        print(f"Error scanning {ip}: {e}")
        return "Unknown", "Unknown"

# Function to scan a single IP address
def scan_ip(ip):
    ping_ip(ip)
    mac = get_mac_from_arp(ip)
    hostname = resolve_hostname(ip)
    vendor, os = get_vendor_and_os(ip)
    return {"ip": ip, "mac": mac, "hostname": hostname, "vendor": vendor, "os": os}

# Function to scan multiple IPs in parallel
def scan_ips_parallel(ip_list, max_threads=10):  # Reduced max_threads
    devices = []
    with ThreadPoolExecutor(max_workers=max_threads) as executor:
        futures = {executor.submit(scan_ip, ip): ip for ip in ip_list}
        for future in as_completed(futures):
            result = future.result()
            if result["mac"] != "Unknown" or result["hostname"] != "Unknown":
                devices.append(result)
            else:
                devices.append({"ip": result["ip"], "mac": "Unknown", "hostname": "Unknown", "vendor": "Unknown", "os": "Unknown"})
    
    # Sort results by IP numerically
    devices.sort(key=lambda x: list(map(int, x["ip"].split("."))))
    return devices

# Function to generate a range of IPs
def generate_ip_range(ip_range_str):
    start_ip, end_ip = ip_range_str.split('-')
    base_ip = '.'.join(start_ip.split('.')[:-1])
    start_octet = int(start_ip.split('.')[-1])
    end_octet = int(end_ip.split('.')[-1])
    return [f"{base_ip}.{i}" for i in range(start_octet, end_octet + 1)]

        // Sorting functionality for the table
        let currentSortColumn = null;
        let currentSortOrder = 'asc';

        function sortTable(columnIndex) {
            const rows = Array.from(document.querySelectorAll("table tbody tr"));
            const th = document.querySelectorAll("table th")[columnIndex];
            const isAscending = currentSortColumn === columnIndex && currentSortOrder === 'asc';

            // Toggle sorting direction
            currentSortOrder = isAscending ? 'desc' : 'asc';
            currentSortColumn = columnIndex;

            // Add sorting indicator to the header
            document.querySelectorAll("table th").forEach((header, index) => {
                header.classList.remove('sort-asc', 'sort-desc');
                if (index === columnIndex) {
                    header.classList.add(isAscending ? 'sort-asc' : 'sort-desc');
                }
            });

            // Sort rows based on column index and order
            rows.sort((rowA, rowB) => {
                const cellA = rowA.cells[columnIndex].innerText;
                const cellB = rowB.cells[columnIndex].innerText;

                if (isNaN(cellA) || isNaN(cellB)) {
                    return isAscending
                        ? cellA.localeCompare(cellB)
                        : cellB.localeCompare(cellA);
                } else {
                    return isAscending ? cellA - cellB : cellB - cellA;
                }
            });

            // Append sorted rows back to the table
            const tbody = document.querySelector("table tbody");
            rows.forEach(row => tbody.appendChild(row));
        }

        // Show loading spinner when scanning
        const form = document.getElementById('scanForm');
        form.addEventListener('submit', function() {
            document.getElementById('loading').style.display = 'block';
        });
    </script>
